💡 检测原理

// 使用 HEAD 请求检测（不下载完整图片） async function checkImageUrl(url) { try { const response = await fetch(url, { method: 'HEAD', mode: 'cors' }); if (response.ok) { return { valid: true, status: response.status }; } return { valid: false, status: response.status, error: response.statusText }; } catch (err) { return { valid: false, error: err.message }; } }

⚠️ 检测局限性

URL 类型 检测结果 说明
完整 URL（如 https://cdn.com/a.png） 可靠 HEAD 请求正常工作
相对路径（如 /images/a.png） 需配置 需要 base URL 前缀
带签名 URL（如 ?signature=xxx） 需额外检查 需验证签名过期时间
CORS 受限的 URL 不可靠 浏览器限制导致检测失败

🔧 替换函数实现

/** * 替换 Markdown 中的图片 URL * @param {string} markdown - 原始 Markdown * @param {Map} replacements - URL 前缀映射 */ function replaceMarkdownUrls(markdown, replacements) { let result = markdown; for (const [oldPrefix, newPrefix] of replacements) { // 处理 URL 前缀替换 const urlPattern = new RegExp( '!\\[.*?\\]\\(' + escapeRegex(oldPrefix) + '([^)]*)\\)', 'g' ); result = result.replace( urlPattern, `![$1](${newPrefix}$1)` ); } return result; } function escapeRegex(str) { return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); }

⚠️ 替换注意事项

  • 确保新 URL 指向的图片与原图内容一致
  • 替换前建议备份原文
  • 替换后验证新链接可访问
  • 相对路径需要额外处理 base URL

5. 策略选择指南

🎯 根据场景选择

场景 推荐策略 实现成本 效果

✅ 推荐实践清单

☐ 优先使用相对路径 — 减少迁移时的 URL 更新工作量

☐ 重要图片本地备份 — 防止第三方服务不可用

☐ CI/CD 集成检测 — 自动化发现失效链接

☐ 多级降级策略 — 关键图片配置备用 CDN

☐ 定期巡检 — 预防胜于补救

☐ 使用稳定域名 — 避免频繁更换图床
