并行复制配置

-- MySQL 5.7+ 从库并行重放
SET GLOBAL slave_parallel_workers = 8;  

-- 按逻辑时钟并行（推荐）
SET GLOBAL slave_parallel_type = 'LOGICAL_CLOCK';

-- MySQL 8.0+ WRITESET 并行
SET GLOBAL slave_parallel_type = 'WRITESET';
SET GLOBAL transaction_write_set_extraction = 'XXHASH64';

大事务拆分

-- ❌ 错误：一次性更新百万行
UPDATE orders SET status = 1 
  WHERE created_at '2024-01-01';

-- ✅ 正确：分批小事务
DELIMITER $$ 
CREATE PROCEDURE batch_update()
BEGIN
  DECLARE i INT DEFAULT 0;
  WHILE i < 1000000 DO
    UPDATE orders SET status = 1 
      WHERE id BETWEEN i AND i + 1000;
    COMMIT;
    SET i = i + 1000;
  END WHILE;
END$$
DELIMITER ;

-- 调用
CALL batch_update();

半同步复制

-- 主库安装半同步插件
INSTALL PLUGIN rpl_semi_sync_master 
  SONAME 'semisync_master.so';

-- 从库安装半同步插件
INSTALL PLUGIN rpl_semi_sync_slave 
  SONAME 'semisync_slave.so';

-- 启用半同步
SET GLOBAL rpl_semi_sync_master_enabled = 1;
SET GLOBAL rpl_semi_sync_slave_enabled = 1;

监控告警 SQL

-- 查看从库状态
SHOW SLAVE STATUS\G;

-- 关键字段说明：
-- Seconds_Behind_Master: 从库落后主库的秒数
-- Slave_IO_Running: IO 线程运行状态
-- Slave_SQL_Running: SQL 线程运行状态
-- Relay_Log_Space: Relay Log 占用的空间

-- 定位正在执行的大事务
SELECT * FROM information_schema.PROCESSLIST
WHERE COMMAND = 'Binlog Dump' 
   OR INFO LIKE '%UPDATE%large%';

关键监控指标

指标正常值告警阈值含义
Seconds_Behind_Master 0 ~ 5s > 30s 从库落后主库的秒数
Slave_IO_Running Yes No IO 线程是否运行
Slave_SQL_Running Yes No SQL 线程是否运行
Relay_Log_Space 稳定 持续增长 Relay Log 磁盘使用
Slave_Open_Temp_Tables 0 > 0 临时表数量（影响复制）
